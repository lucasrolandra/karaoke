// Estados de navegação do App
enum class Screen {
    Search, Player, Score
}

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            MaterialTheme {
                KaraokeTvScreen()
            }
        }
    }

@Composable
fun KaraokeTvScreen() {
    val context = LocalContext.current
    var currentScreen by remember { mutableStateOf(Screen.Search) }
    var videoUrlToPlay by remember { mutableStateOf("") }
    var ultimaNota by remember { mutableStateOf(0) }
    var mensagemNota by remember { mutableStateOf("") }

    // Cores escuras ideais para ambiente de TV
    val backgroundColor = Color(0xFF121214)
    val primaryColor = Color(0xFFE91E63) // Rosa vibrante para os botões

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(backgroundColor),
        contentAlignment = Alignment.Center
    ) {
        when (currentScreen) {
            Screen.Search -> {
                SearchScreen(
                    primaryColor = primaryColor,
                    onMusicFound = { url ->
                        videoUrlToPlay = url
                        Toast.makeText(context, "Carregando vídeo...", Toast.LENGTH_SHORT).show()
                        currentScreen = Screen.Player
                    },
                    onMusicNotFound = {
                        Toast.makeText(context, "Música não encontrada!", Toast.LENGTH_LONG).show()
                    }
                )
            }
            Screen.Player -> {
                // Captura o botão "Voltar" (Esc no emulador ou controle da TV) para fechar o player
                BackHandler {
                    currentScreen = Screen.Search
                }

                VideoPlayerScreen(
                    videoUrl = videoUrlToPlay,
                    onVideoFinished = {
                        // Sorteia uma nota de 70 a 100 ao acabar o vídeo
                        ultimaNota = (70..100).random()
                        mensagemNota = when {
                            ultimaNota >= 95 -> "Incrível! Você é um popstar! 🌟🎤"
                            ultimaNota >= 85 -> "Show de bola! Cantou muito! 🏆"
                            else -> "Mandou bem! Mas ainda pode melhorar. 🎤"
                        }
                        currentScreen = Screen.Score
                    }
                )
            }
            Screen.Score -> {
                // Captura o botão "Voltar" na tela de nota
                BackHandler {
                    currentScreen = Screen.Search
                }

                ScoreScreen(
                    nota = ultimaNota,
                    mensagem = mensagemNota,
                    primaryColor = primaryColor,
                    onBackToSearch = {
                        currentScreen = Screen.Search
                    }
                )
            }
        }
    }
}

@Composable
fun SearchScreen(
    primaryColor: Color,
    onMusicFound: (String) -> Unit,
    onMusicNotFound: () -> Unit
) {
    val context = LocalContext.current
    var codigoMusica by remember { mutableStateOf("") }
    val focusRequester = remember { FocusRequester() }

    // Garante foco automático no campo de texto para uso com controle remoto
    LaunchedEffect(Unit) {
        focusRequester.requestFocus()
    }

    fun executarBusca() {
        if (codigoMusica.isNotBlank()) {
            val url = buscarVideoPorNumero(context, codigoMusica)
            if (url != null) {
                onMusicFound(url)
                codigoMusica = "" // Limpa o campo para a próxima busca
            } else {
                onMusicNotFound()
            }
        }
    }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center,
        modifier = Modifier.padding(24.dp)
    ) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            Text(
                text = "KARAOKÊ BOX",
                color = Color.White,
                fontSize = 42.sp,
                fontWeight = FontWeight.Bold
            )
            Spacer(modifier = Modifier.width(12.dp))
            Text(text = "🎤", fontSize = 38.sp)
        }

        Spacer(modifier = Modifier.height(40.dp))

        OutlinedTextField(
            value = codigoMusica,
            onValueChange = { codigoMusica = it },
            label = { Text("Digite o Código da Música (ex: 1020)", color = Color.Gray) },
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = primaryColor,
                unfocusedBorderColor = Color.Gray,
                focusedLabelColor = primaryColor,
                focusedTextColor = Color.White,
                unfocusedTextColor = Color.White
            ),
            keyboardOptions = KeyboardOptions(
                keyboardType = KeyboardType.Number,
                imeAction = ImeAction.Go
            ),
            keyboardActions = KeyboardActions(
                onGo = { executarBusca() }
            ),
            modifier = Modifier
                .fillMaxWidth(0.8f)
                .focusRequester(focusRequester)
        )

        Spacer(modifier = Modifier.height(24.dp))

        Button(
            onClick = { executarBusca() },
            colors = ButtonDefaults.buttonColors(containerColor = primaryColor),
            modifier = Modifier
                .fillMaxWidth(0.5f)
                .height(48.dp)
        ) {
            Text("Cantar!", fontSize = 18.sp, color = Color.White)
        }
    }
}

@Composable
fun VideoPlayerScreen(videoUrl: String, onVideoFinished: () -> Unit) {
    // Mantém a instância do VideoView salva para podermos pará-lo com segurança no descarte
    var videoViewInstance by remember { mutableStateOf<VideoView?>(null) }

    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
        AndroidView(
            modifier = Modifier.fillMaxSize(),
            factory = { context ->
                VideoView(context).apply {
                    setVideoPath(videoUrl)
                    
                    val mediaController = android.widget.MediaController(context)
                    mediaController.setAnchorView(this)
                    setMediaController(mediaController)

                    setOnCompletionListener {
                        onVideoFinished()
                    }

                    setOnErrorListener { _, _, _ ->
                        Toast.makeText(context, "Erro ao carregar o vídeo. Gerando nota...", Toast.LENGTH_LONG).show()
                        onVideoFinished()
                        true
                    }

                    videoViewInstance = this
                    start()
                }
            },
            update = { videoView ->
                videoView.setVideoPath(videoUrl)
                videoView.start()
            }
        )

        // Limpeza absoluta: Garante o encerramento do áudio e vídeo assim que a tela fecha
        DisposableEffect(Unit) {
            onDispose {
                videoViewInstance?.stopPlayback()
            }
        }
    }
}

@Composable
fun ScoreScreen(
    nota: Int,
    mensagem: String,
    primaryColor: Color,
    onBackToSearch: () -> Unit
) {
    val focusRequester = remember { FocusRequester() }

    LaunchedEffect(Unit) {
        focusRequester.requestFocus()
    }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center,
        modifier = Modifier.padding(24.dp)
    ) {
        Text(text = "SUA NOTA:", color = Color.LightGray, fontSize = 24.sp)
        
        Spacer(modifier = Modifier.height(16.dp))
        
        Text(
            text = "$nota",
            color = Color(0xFFFF5722), // Laranja vibrante para destaque da nota
            fontSize = 96.sp,
            fontWeight = FontWeight.Black
        )

        Spacer(modifier = Modifier.height(24.dp))

        Text(
            text = mensagem,
            color = Color.White,
            fontSize = 22.sp,
            fontWeight = FontWeight.Bold,
            modifier = Modifier.padding(horizontal = 16.dp)
        )

        Spacer(modifier = Modifier.height(48.dp))

        Button(
            onClick = onBackToSearch,
            colors = ButtonDefaults.buttonColors(containerColor = primaryColor),
            modifier = Modifier
                .focusRequester(focusRequester)
                .height(48.dp)
        ) {
            Text("Cantar Outra! 🔄", fontSize = 16.sp, color = Color.White)
        }
    }
}

fun buscarVideoPorNumero(context: Context, numeroDigitado: String): String? {
    try {
        val inputStream = context.assets.open("musicas.txt")
        val reader = BufferedReader(InputStreamReader(inputStream))
        var linha: String?
        while (reader.readLine().also { linha = it } != null) {
            val partes = linha!!.split(",")
            if (partes.isNotEmpty() && partes[0].trim() == numeroDigitado.trim()) {
                var urlOriginal = partes[2].trim()
                
                // CONVERSOR AUTOMÁTICO DO DRIVE: Se o link for do Google Drive, adapta para streaming direto
                if (urlOriginal.contains("drive.google.com")) {
                    // Extrai o ID do arquivo localizado após "/d/"
                    val regex = "/d/([^/]+)".toRegex()
                    val matchResult = regex.find(urlOriginal)
                    val fileId = matchResult?.groupValues?.get(1)
                    
                    if (fileId != null) {
                        urlOriginal = "https://docs.google.com/uc?export=download&id=$fileId"
                    }
                }
                
                return urlOriginal
            }
        }
    } catch (e: Exception) {
        e.printStackTrace()
    }
    return null
}
