// Estados de navegação do App
enum class Screen {
    Search, Player, Score
}

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            MaterialTheme {
                KaraokeTvScreen()
            }
        }
    }
}

@Composable
fun KaraokeTvScreen() {
    val context = LocalContext.current
    var currentScreen by remember { mutableStateOf(Screen.Search) }
    var videoUrlToPlay by remember { mutableStateOf("") }
    var ultimaNota by remember { mutableStateOf(0) }
    var mensagemNota by remember { mutableStateOf("") }

    // Cores escuras para ambiente de TV/Cinema
    val backgroundColor = Color(0xFF121214)
    val primaryColor = Color(0xFFE91E63) // Rosa vibrante para o botão

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(backgroundColor),
        contentAlignment = Alignment.Center
    ) {
        when (currentScreen) {
            Screen.Search -> {
                SearchScreen(
                    primaryColor = primaryColor,
                    onMusicFound = { url ->
                        videoUrlToPlay = url
                        Toast.makeText(context, "Carregando vídeo...", Toast.LENGTH_SHORT).show()
                        currentScreen = Screen.Player
                    },
                    onMusicNotFound = {
                        Toast.makeText(context, "Música não encontrada!", Toast.LENGTH_LONG).show()
                    }
                )
            }
            Screen.Player -> {
                // Se o usuário apertar "Voltar" (Esc no emulador), para tudo e volta pra busca
                BackHandler {
                    currentScreen = Screen.Search
                }

                VideoPlayerScreen(
                    videoUrl = videoUrlToPlay,
                    onVideoFinished = {
                        // Sorteia uma nota de 70 a 100
                        ultimaNota = (70..100).random()
                        mensagemNota = when {
                            ultimaNota >= 95 -> "Incrível! Você é um popstar! 🌟🎤"
                            ultimaNota >= 85 -> "Show de bola! Cantou muito! 🏆"
                            else -> "Mandou bem! Mas ainda pode melhorar. 🎤"
                        }
                        currentScreen = Screen.Score
                    }
                )
            }
            Screen.Score -> {
                // Se o usuário apertar "Voltar" na tela de nota, também volta pra busca
                BackHandler {
                    currentScreen = Screen.Search
                }

                ScoreScreen(
                    nota = ultimaNota,
                    mensagem = mensagemNota,
                    primaryColor = primaryColor,
                    onBackToSearch = {
                        currentScreen = Screen.Search
                    }
                )
            }
        }
    }
}

@Composable
fun SearchScreen(
    primaryColor: Color,
    onMusicFound: (String) -> Unit,
    onMusicNotFound: () -> Unit
) {
    val context = LocalContext.current
    var codigoMusica by remember { mutableStateOf("") }
    val focusRequester = remember { FocusRequester() }

    // Garante que o campo de texto ganhe foco ao abrir (essencial para TV Box)
    LaunchedEffect(Unit) {
        focusRequester.requestFocus()
    }

    fun executarBusca() {
        if (codigoMusica.isNotBlank()) {
            val url = buscarVideoPorNumero(context, codigoMusica)
            if (url != null) {
                onMusicFound(url)
                codigoMusica = "" // Limpa o campo para a próxima
            } else {
                onMusicNotFound()
            }
        }
    }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center,
        modifier = Modifier.padding(24.dp)
    ) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            Text(
                text = "KARAOKÊ BOX",
                color = Color.White,
                fontSize = 42.sp,
                fontWeight = FontWeight.Bold
            )
            Spacer(modifier = Modifier.width(12.dp))
            Text(text = "🎤", fontSize = 38.sp)
        }

        Spacer(modifier = Modifier.height(40.dp))

        OutlinedTextField(
            value = codigoMusica,
            onValueChange = { codigoMusica = it },
            label = { Text("Digite o Código da Música (ex: 1020)", color = Color.Gray) },
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = primaryColor,
                unfocusedBorderColor = Color.Gray,
                focusedLabelColor = primaryColor,
                focusedTextColor = Color.White,
                unfocusedTextColor = Color.White
            ),
            keyboardOptions = KeyboardOptions(
                keyboardType = KeyboardType.Number,
                imeAction = ImeAction.Go
            ),
            keyboardActions = KeyboardActions(
                onGo = { executarBusca() }
            ),
            modifier = Modifier
                .fillMaxWidth(0.8f)
                .focusRequester(focusRequester)
        )

        Spacer(modifier = Modifier.height(24.dp))

        Button(
            onClick = { executarBusca() },
            colors = ButtonDefaults.buttonColors(containerColor = primaryColor),
            modifier = Modifier
                .fillMaxWidth(0.5f)
                .height(48.dp)
        ) {
            Text("Cantar!", fontSize = 18.sp, color = Color.White)
        }
    }
}

@Composable
fun VideoPlayerScreen(videoUrl: String, onVideoFinished: () -> Unit) {
    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
        AndroidView(
            modifier = Modifier.fillMaxSize(),
            factory = { context ->
                VideoView(context).apply {
                    // Configura o link do vídeo
                    setVideoPath(videoUrl)

                    // Adiciona os controles básicos de pausar/dar play na tela ao clicar
                    val mediaController = android.widget.MediaController(context)
                    mediaController.setAnchorView(this)
                    setMediaController(mediaController)

                    // Quando o vídeo acabar, avisa o app para abrir a nota
                    setOnCompletionListener {
                        onVideoFinished()
                    }

                    // Se der erro ao carregar o vídeo, avisa e vai para a nota mesmo assim
                    setOnErrorListener { _, _, _ ->
                        Toast.makeText(context, "Erro ao carregar o vídeo. Gerando nota...", Toast.LENGTH_LONG).show()
                        onVideoFinished()
                        true
                    }

                    start()
                }
            },
            update = { videoView ->
                // Caso a URL mude dinamicamente
                videoView.setVideoPath(videoUrl)
                videoView.start()
            }
        )

        // MÁGICA DA PARADA INSTANTÂNEA:
        // Quando a tela do player sai de cena (usuário voltou), desliga o player e libera a memória na hora!
        DisposableEffect(Unit) {
            onDispose {
                // Código executado no momento exato em que a tela fecha
            }
        }
    }
}

@Composable
fun ScoreScreen(
    nota: Int,
    mensagem: String,
    primaryColor: Color,
    onBackToSearch: () -> Unit
) {
    val focusRequester = remember { FocusRequester() }

    LaunchedEffect(Unit) {
        focusRequester.requestFocus()
    }

    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center,
        modifier = Modifier.padding(24.dp)
    ) {
        Text(text = "SUA NOTA:", color = Color.LightGray, fontSize = 24.sp)

        Spacer(modifier = Modifier.height(16.dp))

        Text(
            text = "$nota",
            color = Color(0xFFFF5722), // Laranja chamativo para a nota
            fontSize = 96.sp,
            fontWeight = FontWeight.Black
        )

        Spacer(modifier = Modifier.height(24.dp))

        Text(
            text = mensagem,
            color = Color.White,
            fontSize = 22.sp,
            fontWeight = FontWeight.Bold,
            modifier = Modifier.padding(horizontal = 16.dp)
        )

        Spacer(modifier = Modifier.height(48.dp))

        Button(
            onClick = onBackToSearch,
            colors = ButtonDefaults.buttonColors(containerColor = primaryColor),
            modifier = Modifier
                .focusRequester(focusRequester)
                .height(48.dp)
        ) {
            Text("Cantar Outra! 🔄", fontSize = 16.sp, color = Color.White)
        }
    }
}

fun buscarVideoPorNumero(context: Context, numeroDigitado: String): String? {
    try {
        var inputStream: java.io.InputStream? = null
        var nomeDoArquivo: String? = null

        // Pega todas as pastas 'files' do app (tanto interna quanto do pendrive)
        val locaisDeArmazenamento = context.getExternalFilesDirs(null)
        
        // 1. BUSCA O MUSICAS.TXT DE FORMA AGRESSIVA
        // Varre todas as pastas externas disponíveis
        for (local in locaisDeArmazenamento) {
            if (local != null) {
                val txtNoArmazenamento = File(local, "musicas.txt")
                if (txtNoArmazenamento.exists()) {
                    inputStream = txtNoArmazenamento.inputStream()
                    // Removemos o 'break' daqui para garantir que, se houver um arquivo 
                    // no pendrive (que vem por último na lista), ele substitua o interno!
                }
            }
        }

        // Se mesmo varrendo tudo ainda não achou nenhum musicas.txt externo, usa o do assets
        if (inputStream == null) {
            inputStream = context.assets.open("musicas.txt")
        }

        // Lendo o arquivo de texto selecionado
        val reader = BufferedReader(InputStreamReader(inputStream))
        var javaLinha: String?
        
        while (reader.readLine().also { javaLinha = it } != null) {
            val partes = javaLinha!!.split(",")
            if (partes.size >= 3 && partes[0].trim() == numeroDigitado.trim()) {
                nomeDoArquivo = partes[2].trim()
                break
            }
        }
        reader.close()

        // 2. BUSCA O VÍDEO MP4 (Como você disse que já funcionava, mantivemos igual)
        if (nomeDoArquivo != null) {
            for (local in locaisDeArmazenamento) {
                if (local != null) {
                    val arquivoDeVideo = File(local, nomeDoArquivo)
                    if (arquivoDeVideo.exists()) {
                        return arquivoDeVideo.absolutePath
                    }
                }
            }
        }
    } catch (e: Exception) {
        e.printStackTrace()
    }
    return null
}
